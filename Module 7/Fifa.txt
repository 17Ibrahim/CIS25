#include <iostream>
#include <fstream>
#include <sstream>
#include <vector>
using namespace std;

// Class representing one World Cup match
class Match
{
public:
    int matchID;
    string date;
    string team1;
    string team2;
    int score1;
    int score2;
    string winner;

    void display()
    {
        cout << "Match ID: " << matchID << endl;
        cout << "Date: " << date << endl;
        cout << "Teams: " << team1 << " vs " << team2 << endl;
        cout << "Score: " << score1 << " - " << score2 << endl;
        cout << "Winner: " << winner << endl;
        cout << "-----------------------------" << endl;
    }
};

int main()
{
    ifstream file("matches.csv");

    if (!file)
    {
        cout << "Error opening matches.csv" << endl;
        return 1;
    }

    vector<Match> matches;

    string line;

    // Skip the header row
    getline(file, line);

    // Read first 10 matches
    while (getline(file, line) && matches.size() < 10)
    {
        stringstream ss(line);
        Match m;
        string temp;

        // match_id
        getline(ss, temp, ',');
        m.matchID = stoi(temp);

        // date
        getline(ss, m.date, ',');

        // stage (skip)
        getline(ss, temp, ',');

        // group (skip)
        getline(ss, temp, ',');

        // team_1
        getline(ss, m.team1, ',');

        // team_2
        getline(ss, m.team2, ',');

        // score_1
        getline(ss, temp, ',');
        m.score1 = stoi(temp);

        // score_2
        getline(ss, temp, ',');
        m.score2 = stoi(temp);

        // winner
        getline(ss, m.winner, ',');

        matches.push_back(m);
    }

    file.close();

    cout << "2026 FIFA World Cup Matches\n";
    cout << "===========================\n\n";

    for (Match m : matches)
    {
        m.display();
    }

    return 0;
}
