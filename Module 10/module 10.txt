#include <iostream>
#include <string>
using namespace std;

class Product {
private:
    int id;
    string name;
    float price;

public:
    // Constructor
    Product(int i, string n, float p) {
        id = i;
        name = n;
        price = p;
        cout << "Constructor called\n";
    }

    // Destructor
    ~Product() {
        cout << "Destructor called\n";
    }

    void printDetails() {
        cout << "ID: " << id << ", Name: " << name << ", Price: $" << price << endl;
    }
};

int main() {
    Product p(101, "Notebook", 4.99);
    p.printDetails();
    return 0;
}

//OUTPUT

Constructor called
ID: 101, Name: Notebook, Price: $4.99
Destructor called

//Knowledge Check

1. Using constructors and destructors:
Think of constructors as the setup crew. In this project, you use them to instantly assign the id, name, and price the moment a Product is created. Destructors are the cleanup crew. You use them to handle any memory management or logging when the product object is no longer needed.

2. Similarities:
The constructor and the destructor share the exact same name as the class itself (in this case, Product). Another big shared trait is that neither of them has a return type—not even void.
